program DocumentationExample;
uses Std.Tui as Tui;
type TuiStyleRole = Tui.TuiStyleRole;
type PaletteColor = record Green: integer; end record;
type PaletteStyle = record Foreground: PaletteColor; end record;
type Palette = record
  function ForRole(Self: Palette; Role: TuiStyleRole): PaletteStyle;
  begin return PaletteStyle(Foreground := PaletteColor(Green := 128)); end function;
end record;
function BuildPalette(): Palette; begin return Palette(); end function;
function LoadItems(): array of string; begin return ['first']; end function;
type Num = record
  Value: integer;
  static function Create(Value: integer): Num; begin return Num(Value := Value); end function;
  function Scale(Self: Num; Factor: integer): Num; begin return Num(Value := Self.Value * Factor); end function;
  function Next(Self: Num): Num; begin return Num(Value := Self.Value + 1); end function;
end record;
begin
// @example
end.
