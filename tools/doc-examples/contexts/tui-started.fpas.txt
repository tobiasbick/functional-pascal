program DocumentationExample;
uses Std.Ui.Terminal as Tui, Std.Tasks as Tasks;
type TuiCmd = Tui.TuiCmd;
type TuiCmdOutput = Tui.TuiCmdOutput;
type TuiMsg = Tui.TuiMsg;
function LoadData(Token: Tasks.CancellationToken): result of (boolean, string); begin return Ok(true); end function;
procedure Update(Msg: TuiMsg; Cmd: TuiCmdOutput);
begin
case Msg of
// @example
when TuiMsg.QuitRequested, TuiMsg.Key(_), TuiMsg.Pointer(_), TuiMsg.Tick(_), TuiMsg.Resize(_), TuiMsg.FocusChanged(_, _), TuiMsg.Action(_, _), TuiMsg.TextChanged(_, _, _, _), TuiMsg.TextAreaChanged(_, _, _, _, _), TuiMsg.CheckChanged(_, _, _), TuiMsg.SelectionChanged(_, _, _), TuiMsg.ScrollChanged(_, _, _), TuiMsg.MenuChanged(_, _, _), TuiMsg.BackgroundFailed(_, _, _, _, _, _): null;
end case;
end procedure;
begin end.
